# //! Windows 原生守护与隔离测试；不加载输入引擎或翻译运行时。
cmake_minimum_required(VERSION 3.20)
project(QingjianWatchdog LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Watchdog requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
function(configure_guard target)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
  target_link_libraries(${target} PRIVATE advapi32 shell32 user32)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    target_link_options(${target} PRIVATE -municode -static)
  endif()
endfunction()
add_executable(qingjian-watchdog WIN32 Watchdog.cpp)
configure_guard(qingjian-watchdog)
include(CTest)
if(BUILD_TESTING)
  add_executable(watchdog-test WIN32 Watchdog.cpp)
  target_compile_definitions(watchdog-test PRIVATE WATCHDOG_TEST)
  configure_guard(watchdog-test)
  add_executable(fixture WIN32 tests/Fixture.cpp)
  configure_guard(fixture)
  add_test(NAME watchdog-isolated COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/Test-Watchdog.ps1" -BinDir "$<TARGET_FILE_DIR:watchdog-test>"
    -WorkDir "${CMAKE_CURRENT_BINARY_DIR}/integration")
  add_test(NAME watchdog-task COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/Test-Task.ps1" -BinDir "$<TARGET_FILE_DIR:watchdog-test>"
    -WorkDir "${CMAKE_CURRENT_BINARY_DIR}/task-integration")
  set_tests_properties(watchdog-isolated watchdog-task PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
endif()
